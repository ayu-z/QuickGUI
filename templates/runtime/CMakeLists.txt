cmake_minimum_required(VERSION 3.20)
project(QuickGUIExport LANGUAGES C CXX)
find_program(NPM_EXECUTABLE NAMES npm npm.cmd REQUIRED)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
add_custom_target(web COMMAND ${NPM_EXECUTABLE} run build WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_custom_target(engine COMMAND ${CARGO_EXECUTABLE} build --release -p quickgui-engine -p quickgui-c-runner WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_custom_target(desktop COMMAND ${NPM_EXECUTABLE} run desktop:build WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
if(EXISTS "${CMAKE_SOURCE_DIR}/sdk/aot/CMakeLists.txt")
  add_subdirectory(sdk/aot)
endif()
