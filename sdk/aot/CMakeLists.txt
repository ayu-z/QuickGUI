cmake_minimum_required(VERSION 3.20)
project(QuickGUIScript LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT DEFINED QG_SCRIPT_SOURCE)
  set(QG_SCRIPT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/example.c")
endif()
add_library(quickgui_script STATIC "${QG_SCRIPT_SOURCE}")
target_include_directories(quickgui_script PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
set_target_properties(quickgui_script PROPERTIES POSITION_INDEPENDENT_CODE ON
  ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
  ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/lib"
  ARCHIVE_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/lib")
